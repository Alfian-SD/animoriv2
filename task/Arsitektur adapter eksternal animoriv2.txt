Arsitektur adapter eksternal **animoriv2** memisahkan logika ekstraksi aliran video dari *adapter* katalog sehingga kegagalan atau perubahan struktur DOM pada satu sumber tidak memengaruhi ketersediaan `curatedProvider` maupun jalur *streaming* utama. Tambahkan dependensi `cheerio` (`npm install cheerio`) untuk pemrosesan HTML sisi server dengan dukungan tipe TypeScript bawaan.

## 1. Stream Extractor & Dean Edwards Unpacker (`src/server/extractors/streamExtractor.ts`)

Modul ini mendekode skrip *embed player* yang dikemas menggunakan pola `eval(function(p,a,c,k,e,d)...)`, mengekstrak URL langsung (`.m3u8` atau `.mp4`), serta memvalidasi hasilnya melalui `isSafeAllowedStreamUrl` sebelum diteruskan ke klien.

```typescript
// src/server/extractors/streamExtractor.ts
import * as cheerio from "cheerio";
import type { StreamMode } from "../../shared/anime.js";
import { SERVER_CONFIG } from "../config.js";
import { isSafeAllowedStreamUrl } from "../security/ssrfGuard.js";

export interface ExtractedStream {
  url: string;
  mode: Extract<StreamMode, "hls" | "mp4">;
}

const BASE62_ALPHABET =
  "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";

function encodeRadix(num: number, radix: number): string {
  const prefix = num < radix ? "" : encodeRadix(Math.floor(num / radix), radix);
  const remainder = num % radix;
  const digit =
    remainder > 35
      ? String.fromCharCode(remainder + 29)
      : (BASE62_ALPHABET[remainder] ?? "");
  return prefix + digit;
}

/**
 * Mendekode skrip terkompresi pola Dean Edwards P.A.C.K.E.R.
 */
export function unpackDeanEdwards(packedHtml: string): string {
  const match = packedHtml.match(
    /\}\('(.+?)',(\d+),(\d+),'(.+?)'\.split\('\|'\)/s
  );
  if (!match || !match[1] || !match[2] || !match[3] || !match[4]) {
    return packedHtml;
  }

  let payload = match[1];
  const radix = Number.parseInt(match[2], 10);
  let count = Number.parseInt(match[3], 10);
  const words = match[4].split("|");

  if (!Number.isFinite(radix) || radix < 2 || radix > 62 || !Number.isFinite(count)) {
    return packedHtml;
  }

  while (count > 0) {
    count -= 1;
    const token = words[count];
    if (token) {
      const encodedKey = encodeRadix(count, radix);
      payload = payload.replace(new RegExp(`\\b${encodedKey}\\b`, "g"), token);
    }
  }

  return payload;
}

/**
 * Mengekstrak URL stream langsung (.m3u8 atau .mp4) dari halaman iframe/player.
 */
export async function extractDirectStream(
  playerUrl: string,
  referer: string
): Promise<ExtractedStream | null> {
  if (!isSafeAllowedStreamUrl(playerUrl)) {
    return null;
  }

  try {
    const response = await fetch(playerUrl, {
      headers: {
        "User-Agent": SERVER_CONFIG.chromeUserAgent,
        Referer: referer,
      },
      signal: AbortSignal.timeout(SERVER_CONFIG.metadataTimeoutMs),
    });

    if (!response.ok) return null;

    const rawHtml = await response.text();
    const unpackedHtml = unpackDeanEdwards(rawHtml);
    const $ = cheerio.load(unpackedHtml);

    let candidate =
      $("video source").attr("src") ?? $("video").attr("src") ?? null;

    if (!candidate) {
      const hlsMatch = unpackedHtml.match(
        /(?:file|src|url)\s*[:=]\s*["'](https?:\/\/[^"']+\.m3u8[^"']*)["']/i
      );
      const mp4Match = unpackedHtml.match(
        /(?:videoURL|file|src|url)\s*[:=]\s*["'](https?:\/\/[^"']+\.mp4[^"']*)["']/i
      );
      candidate = hlsMatch?.[1] ?? mp4Match?.[1] ?? null;
    }

    if (!candidate) return null;

    const cleanUrl = candidate.replace(/&amp;/g, "&").trim();
    if (!isSafeAllowedStreamUrl(cleanUrl)) {
      return null;
    }

    return {
      url: cleanUrl,
      mode: cleanUrl.toLowerCase().includes(".m3u8") ? "hls" : "mp4",
    };
  } catch {
    return null;
  }
}
```

## 2. Adapter Provider Eksternal & AJAX Mirror Resolver (`src/server/providers/externalProvider.ts`)

Adapter ini menggabungkan tiga mekanisme utama:

1. **Normalisasi Slug Tanpa Domain Keras:** Menghapus protokol dan host sehingga perubahan TLD sumber eksternal tidak merusak rute internal.
2. **Ekstraksi Hydrated Script State:** Memindai muatan JSON di dalam skrip hidrasi halaman modern.
3. **Resolusi Nonce WordPress AJAX Konkuren:** Mengambil *nonce* sekali per episode lalu menyelesaikan seluruh *mirror* secara paralel melalui `Promise.allSettled`, mengurutkan kualitas dari `1080p` ke `360p`, serta menyimpan hasil resolusi dalam *TTL cache* singkat (15 menit) agar token CDN bertanda waktu tidak kedaluwarsa.

```typescript
// src/server/providers/externalProvider.ts
import * as cheerio from "cheerio";
import type {
  Anime,
  PlaybackOption,
  VideoQuality,
} from "../../shared/anime.js";
import { SERVER_CONFIG } from "../config.js";
import { isSafeAllowedStreamUrl } from "../security/ssrfGuard.js";
import { extractDirectStream } from "../extractors/streamExtractor.js";

export interface CatalogProvider {
  readonly id: string;
  fetchCatalog(): Promise<Anime[]>;
  resolveEpisodeMirrors(episodeUrl: string): Promise<PlaybackOption[]>;
}

export interface HydratedStreamEntry {
  playUrl: string;
  quality: VideoQuality;
  episode: number;
}

interface MirrorTask {
  quality: VideoQuality;
  serverName: string;
  encodedPayload: string;
}

const QUALITY_WEIGHT: Record<VideoQuality, number> = {
  "1080p": 4,
  "720p": 3,
  "480p": 2,
  "360p": 1,
};

function normalizeQuality(raw: string): VideoQuality {
  if (raw.includes("1080")) return "1080p";
  if (raw.includes("720")) return "720p";
  if (raw.includes("360")) return "360p";
  return "480p";
}

export function extractSlugFromHref(rawHref = ""): string {
  return (
    rawHref
      .replace(/^https?:\/\/[^/]+/i, "")
      .replace(/^\/+|\/+$/g, "")
      .split("/")
      .pop() ?? ""
  );
}

/**
 * Mengekstrak objek stream dari skrip hidrasi Next.js / React Flight.
 */
export function parseHydratedStreams(html: string): HydratedStreamEntry[] {
  const results: HydratedStreamEntry[] = [];
  const regex = /\{[^{}]*?\\"playUrl\\"\s*:\s*\\"([^"\\]+)\\"[^{}]*?\}/g;
  let match: RegExpExecArray | null;

  while ((match = regex.exec(html)) !== null) {
    const rawBlock = match[0];
    try {
      const unescaped = rawBlock.replace(/\\"/g, '"').replace(/\\\\/g, "\\");
      const parsed = JSON.parse(unescaped) as Record<string, unknown>;
      const playUrl = typeof parsed.playUrl === "string" ? parsed.playUrl : "";
      if (playUrl && isSafeAllowedStreamUrl(playUrl)) {
        results.push({
          playUrl,
          quality: normalizeQuality(String(parsed.quality ?? "720p")),
          episode: Number(parsed.episode ?? 1) || 1,
        });
      }
    } catch {
      const urlMatch = rawBlock.match(/\\"playUrl\\":\\"([^"\\]+)\\"/);
      const qualMatch = rawBlock.match(/\\"quality\\":\\"([^"\\]+)\\"/);
      const epMatch = rawBlock.match(/\\"episode\\":(\d+)/);
      const playUrl = urlMatch?.[1] ?? "";
      if (playUrl && isSafeAllowedStreamUrl(playUrl)) {
        results.push({
          playUrl,
          quality: normalizeQuality(qualMatch?.[1] ?? "720p"),
          episode: Number.parseInt(epMatch?.[1] ?? "1", 10) || 1,
        });
      }
    }
  }

  return results;
}

export function sortPlaybackOptionsByQuality(
  options: PlaybackOption[]
): PlaybackOption[] {
  return [...options].sort(
    (a, b) => (QUALITY_WEIGHT[b.quality] ?? 0) - (QUALITY_WEIGHT[a.quality] ?? 0)
  );
}

export class ExternalHtmlScraperProvider implements CatalogProvider {
  public readonly id: string;
  private readonly baseUrl: string;
  private readonly mirrorCache = new Map<
    string,
    { options: PlaybackOption[]; expiresAt: number }
  >();
  private static readonly MIRROR_TTL_MS = 15 * 60 * 1000; // 15 menit (mencegah token CDN kedaluwarsa)

  constructor(id: string, baseUrl: string) {
    this.id = id;
    this.baseUrl = baseUrl.replace(/\/+$/, "");
  }

  public async fetchCatalog(): Promise<Anime[]> {
    if (!this.baseUrl || !isSafeAllowedStreamUrl(this.baseUrl)) {
      return [];
    }

    try {
      const res = await fetch(`${this.baseUrl}/`, {
        headers: {
          "User-Agent": SERVER_CONFIG.chromeUserAgent,
          Referer: `${this.baseUrl}/`,
        },
        signal: AbortSignal.timeout(SERVER_CONFIG.metadataTimeoutMs),
      });

      if (!res.ok) return [];
      const html = await res.text();
      const $ = cheerio.load(html);
      const items: Anime[] = [];

      $(".venz ul li").each((index, el) => {
        const title = $(el).find("h2.jdlflm").text().trim();
        const href = $(el).find("a").attr("href") ?? "";
        const posterUrl = $(el).find("img").attr("src");
        const slug = extractSlugFromHref(href);

        if (!title || !slug) return;

        items.push({
          id: `${this.id}:${slug}`,
          slug,
          sourceProvider: this.id,
          title: { romaji: title },
          synopsis: `Katalog sinkronisasi eksternal dari ${this.id}.`,
          status: "RELEASING",
          seasonYear: 2026,
          season: "FALL",
          score: 8.0,
          genres: ["Action", "Adventure"],
          studio: "External Source",
          artwork: {
            motif: index % 2 === 0 ? "ridgelines" : "sun",
            primaryHue: (index * 47) % 360,
            posterUrl:
              posterUrl && isSafeAllowedStreamUrl(posterUrl)
                ? posterUrl
                : undefined,
          },
          episodes: [],
        });
      });

      return items;
    } catch {
      return [];
    }
  }

  public async resolveEpisodeMirrors(
    episodeUrl: string
  ): Promise<PlaybackOption[]> {
    if (!isSafeAllowedStreamUrl(episodeUrl)) {
      return [];
    }

    const cached = this.mirrorCache.get(episodeUrl);
    if (cached && cached.expiresAt > Date.now()) {
      return cached.options;
    }

    try {
      const epRes = await fetch(episodeUrl, {
        headers: {
          "User-Agent": SERVER_CONFIG.chromeUserAgent,
          Referer: `${this.baseUrl}/`,
        },
        signal: AbortSignal.timeout(SERVER_CONFIG.metadataTimeoutMs),
      });

      if (!epRes.ok) return [];
      const episodeHtml = await epRes.text();

      const actionMatches = [
        ...episodeHtml.matchAll(/action:\s*["']([a-f0-9]{32})["']/g),
      ].map((m) => m[1]);

      const streamAction = actionMatches[0];
      const nonceAction = actionMatches[1];
      if (!streamAction || !nonceAction) {
        return [];
      }

      const ajaxEndpoint = `${this.baseUrl}/wp-admin/admin-ajax.php`;
      const nonceRes = await fetch(ajaxEndpoint, {
        method: "POST",
        headers: {
          "Content-Type": "application/x-www-form-urlencoded",
          "User-Agent": SERVER_CONFIG.chromeUserAgent,
          Referer: episodeUrl,
        },
        body: new URLSearchParams({ action: nonceAction }),
        signal: AbortSignal.timeout(SERVER_CONFIG.metadataTimeoutMs),
      });

      if (!nonceRes.ok) return [];
      const nonceJson = (await nonceRes.json()) as { data?: string };
      const nonce = nonceJson.data;
      if (!nonce) return [];

      const $ = cheerio.load(episodeHtml);
      const tasks: MirrorTask[] = [];

      $(".mirrorstream ul").each((_, ul) => {
        const rawClass = $(ul).attr("class") ?? "480p";
        const quality = normalizeQuality(rawClass);

        $(ul)
          .find("li a[data-content]")
          .each((__, anchor) => {
            const serverName = $(anchor).text().trim();
            const encodedPayload = $(anchor).attr("data-content") ?? "";
            if (serverName && encodedPayload) {
              tasks.push({ quality, serverName, encodedPayload });
            }
          });
      });

      const settled = await Promise.allSettled(
        tasks.map(async (task, idx): Promise<PlaybackOption | null> => {
          const decodedJson = Buffer.from(
            task.encodedPayload,
            "base64"
          ).toString("utf8");
          const payloadObj = JSON.parse(decodedJson) as Record<string, string>;

          const mirrorRes = await fetch(ajaxEndpoint, {
            method: "POST",
            headers: {
              "Content-Type": "application/x-www-form-urlencoded",
              "User-Agent": SERVER_CONFIG.chromeUserAgent,
              Referer: episodeUrl,
            },
            body: new URLSearchParams({
              ...payloadObj,
              nonce,
              action: streamAction,
            }),
            signal: AbortSignal.timeout(SERVER_CONFIG.metadataTimeoutMs),
          });

          if (!mirrorRes.ok) return null;
          const mirrorJson = (await mirrorRes.json()) as { data?: string };
          if (!mirrorJson.data) return null;

          const iframeHtml = Buffer.from(mirrorJson.data, "base64").toString(
            "utf8"
          );
          const iframeSrc =
            cheerio.load(iframeHtml)("iframe").attr("src") ?? "";
          if (!iframeSrc || !isSafeAllowedStreamUrl(iframeSrc)) {
            return null;
          }

          const direct = await extractDirectStream(iframeSrc, episodeUrl);

          return {
            id: `${this.id}-${task.quality}-${idx}`,
            providerId: this.id,
            serverName: task.serverName,
            quality: task.quality,
            mode: direct ? direct.mode : "embed",
            streamUrl: direct ? direct.url : iframeSrc,
          };
        })
      );

      const validOptions = settled
        .filter(
          (r): r is PromiseFulfilledResult<PlaybackOption> =>
            r.status === "fulfilled" && r.value !== null
        )
        .map((r) => r.value);

      const sorted = sortPlaybackOptionsByQuality(validOptions);
      this.mirrorCache.set(episodeUrl, {
        options: sorted,
        expiresAt: Date.now() + ExternalHtmlScraperProvider.MIRROR_TTL_MS,
      });

      return sorted;
    } catch {
      return [];
    }
  }
}
```

## 3. Pembaruan Multi-Source Aggregator (`src/server/services/aggregator.ts`)

Perbarui `CatalogAggregator` agar dapat menggabungkan `CuratedProvider` (prioritas pertama untuk metadata editorial) dengan beberapa `CatalogProvider` eksternal secara paralel menggunakan `Promise.allSettled`, sekaligus mempertahankan peta rute $\mathcal{O}(1)$ (`sourceRouteMap`).

```typescript
// src/server/services/aggregator.ts
import type { Anime } from "../../shared/anime.js";
import type { CatalogProvider } from "../providers/externalProvider.js";

export interface RouteEntry {
  providerId: string;
  animeId: string;
  updatedAt: number;
}

export function normalizeTitle(rawTitle = ""): string {
  return rawTitle
    .toLowerCase()
    .replace(
      /\s*(sub(?:title)?\s*indo(?:nesia)?|bd|batch|ova|ona|s\d+|season\s*\d+)\b.*$/gi,
      ""
    )
    .replace(/[^a-z0-9]+/g, " ")
    .trim();
}

export class CatalogAggregator {
  private readonly sourceRouteMap = new Map<string, RouteEntry>();
  private readonly providers = new Map<string, CatalogProvider>();
  private cachedCatalog: Anime[] = [];
  private cacheExpiresAt = 0;
  private static readonly CATALOG_TTL_MS = 5 * 60 * 1000; // 5 menit

  public registerProvider(provider: CatalogProvider): void {
    this.providers.set(provider.id, provider);
  }

  public registerCatalog(items: Anime[]): Anime[] {
    const seenNormalizedTitles = new Set<string>();
    const deduplicated: Anime[] = [];

    for (const anime of items) {
      const now = Date.now();
      const entry: RouteEntry = {
        providerId: anime.sourceProvider,
        animeId: anime.id,
        updatedAt: now,
      };
      this.sourceRouteMap.set(anime.id, entry);
      this.sourceRouteMap.set(anime.slug, entry);

      const normKey = normalizeTitle(anime.title.romaji);
      if (!seenNormalizedTitles.has(normKey)) {
        seenNormalizedTitles.add(normKey);
        deduplicated.push(anime);
      }
    }

    return deduplicated;
  }

  public async syncAllProviders(
    curatedItems: Anime[],
    forceRefresh = false
  ): Promise<Anime[]> {
    if (
      !forceRefresh &&
      this.cachedCatalog.length > 0 &&
      this.cacheExpiresAt > Date.now()
    ) {
      return this.cachedCatalog;
    }

    const externalList = [...this.providers.values()];
    const settled = await Promise.allSettled(
      externalList.map((provider) => provider.fetchCatalog())
    );

    const externalItems: Anime[] = [];
    for (const res of settled) {
      if (res.status === "fulfilled") {
        externalItems.push(...res.value);
      }
    }

    // Prioritaskan curatedItems terlebih dahulu agar metadata editorial tetap utuh
    const merged = this.registerCatalog([...curatedItems, ...externalItems]);
    this.cachedCatalog = merged;
    this.cacheExpiresAt = Date.now() + CatalogAggregator.CATALOG_TTL_MS;

    return merged;
  }

  public resolveRoute(idOrSlug: string): RouteEntry | null {
    return this.sourceRouteMap.get(idOrSlug) ?? null;
  }

  public getProvider(providerId: string): CatalogProvider | undefined {
    return this.providers.get(providerId);
  }

  public size(): number {
    return this.sourceRouteMap.size;
  }
}

```

## 4. Unit Test Ekstraktor & Mirror (`tests/extractor.test.ts`)

Tambahkan pengujian Vitest untuk memverifikasi *unpacker* Dean Edwards, ekstraksi skrip hidrasi, blokir SSRF pada URL hasil ekstraksi, dan pengurutan kualitas *mirror*.

```typescript
// tests/extractor.test.ts
import { describe, expect, it } from "vitest";
import { unpackDeanEdwards } from "../src/server/extractors/streamExtractor.js";
import {
  extractSlugFromHref,
  parseHydratedStreams,
  sortPlaybackOptionsByQuality,
} from "../src/server/providers/externalProvider.js";
import type { PlaybackOption } from "../src/shared/anime.js";

describe("External Scraper & Stream Extractor", () => {
  it("unpacks Dean Edwards P.A.C.K.E.R. obfuscated player config", () => {
    const packed =
      "eval(function(p,a,c,k,e,d){return p}('0 1=\"2://3.4/5.6\";',7,7,'var|streamUrl|https|cdn|example|video|m3u8'.split('|'),0,{}))";
    const unpacked = unpackDeanEdwards(packed);
    expect(unpacked).toContain('var streamUrl="https://cdn.example/video.m3u8";');
  });

  it("extracts valid public streams from hydrated Next.js payloads and drops SSRF URLs", () => {
    const html = `
      <script>
        self.__next_f.push([1, "{\\"playUrl\\":\\"https://cdn.example.com/ep1.m3u8\\",\\"quality\\":\\"1080p\\",\\"episode\\":1}"]);
        self.__next_f.push([1, "{\\"playUrl\\":\\"http://169.254.169.254/latest/meta-data/\\",\\"quality\\":\\"720p\\",\\"episode\\":1}"]);
      </script>
    `;
    const streams = parseHydratedStreams(html);
    expect(streams).toHaveLength(1);
    expect(streams[0]).toEqual({
      playUrl: "https://cdn.example.com/ep1.m3u8",
      quality: "1080p",
      episode: 1,
    });
  });

  it("normalizes slugs independent of upstream TLD rotations", () => {
    expect(
      extractSlugFromHref("https://otakudesu.cloud/anime/frieren-sub-indo/")
    ).toBe("frieren-sub-indo");
    expect(
      extractSlugFromHref("https://otakudesu.best/anime/frieren-sub-indo")
    ).toBe("frieren-sub-indo");
  });

  it("sorts resolved playback mirrors from highest to lowest resolution", () => {
    const unsorted: PlaybackOption[] = [
      {
        id: "1",
        providerId: "ext",
        serverName: "Mirror A",
        quality: "480p",
        mode: "embed",
        streamUrl: "https://cdn.example.com/480",
      },
      {
        id: "2",
        providerId: "ext",
        serverName: "Mirror B",
        quality: "1080p",
        mode: "hls",
        streamUrl: "https://cdn.example.com/1080.m3u8",
      },
      {
        id: "3",
        providerId: "ext",
        serverName: "Mirror C",
        quality: "720p",
        mode: "mp4",
        streamUrl: "https://cdn.example.com/720.mp4",
      },
    ];

    const sorted = sortPlaybackOptionsByQuality(unsorted);
    expect(sorted.map((o) => o.quality)).toEqual(["1080p", "720p", "480p"]);
  });
});
```
